import type { Metadata } from "next";
import Link from "next/link";
import { notFound } from "next/navigation";
import {
  ArrowLeft,
  ArrowRight,
  CalendarCheck,
  Check,
  Clock,
  Music4,
  Sparkles,
  Users,
  Wrench,
} from "lucide-react";
import { getClubSettings } from "@/lib/services/club-settings";
import {
  getRoomBySlug,
  listBookableRooms,
  loadAvailabilityContext,
  nextAvailableSlots,
  todayInClubTime,
} from "@/lib/services/availability";
import { RoomGallery } from "@/components/site/room-gallery";
import { RoomCard, categoryLabel } from "@/components/site/room-card";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { formatMoney } from "@/lib/utils/money";
import { formatClubDateLong, formatClubTime12h } from "@/lib/utils/datetime";
import { DAY_LABELS, formatOperatingDays } from "@/lib/utils/opening-hours";

export const dynamic = "force-dynamic";

export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  const { slug } = await params;
  const room = await getRoomBySlug(slug).catch(() => null);
  if (!room) return { title: "Room not found" };
  return {
    title: `${room.name} — ${categoryLabel(room.categorySlug)}`,
    description:
      room.shortDescription ??
      `${room.name}: a private ${categoryLabel(room.categorySlug)} room for up to ${room.capacity} guests at ${formatMoney(room.hourlyRate)} per hour.`,
  };
}

export default async function RoomDetailPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const [room, settings] = await Promise.all([getRoomBySlug(slug), getClubSettings()]);

  if (!room) notFound();

  const today = todayInClubTime(settings);
  const [context, allRooms] = await Promise.all([
    loadAvailabilityContext(settings, today).catch(() => null),
    listBookableRooms().catch(() => []),
  ]);

  const slots = context
    ? nextAvailableSlots(context, room.id, room.minimumDuration, 4)
    : [];
  const similar = allRooms
    .filter((candidate) => candidate.id !== room.id && candidate.categorySlug === room.categorySlug)
    .slice(0, 2);
  const alternatives =
    similar.length > 0 ? similar : allRooms.filter((c) => c.id !== room.id).slice(0, 2);

  const images = room.images.length > 0 ? room.images : ["/images/room-standard.jpg"];

  return (
    <div className="mx-auto w-full max-w-7xl px-5 py-8 sm:px-8 sm:py-12">
      <Link
        href="/rooms"
        className="inline-flex items-center gap-1.5 text-[12.5px] text-muted transition-colors hover:text-foreground"
      >
        <ArrowLeft aria-hidden className="size-3.5" />
        All rooms
      </Link>

      <div className="mt-5 grid gap-8 lg:grid-cols-[1.55fr_1fr] lg:gap-10">
        <div className="min-w-0">
          <RoomGallery images={images} alt={room.name} />

          <header className="mt-7">
            <div className="flex flex-wrap items-center gap-2">
              <Badge variant={room.categorySlug === "standard" ? "neutral" : "gold"}>
                {categoryLabel(room.categorySlug)}
              </Badge>
              {room.maintenanceStatus !== "OPERATIONAL" ? (
                <Badge variant="warning">
                  <Wrench className="size-3" />
                  Under maintenance
                </Badge>
              ) : null}
              {!room.isBookable && room.maintenanceStatus === "OPERATIONAL" ? (
                <Badge variant="danger">Currently unavailable</Badge>
              ) : null}
            </div>

            <h1 className="mt-3 font-display text-3xl leading-tight tracking-tight sm:text-4xl">
              {room.name}
            </h1>

            <ul className="mt-3 flex flex-wrap items-center gap-x-5 gap-y-2 text-[13px] text-muted">
              <li className="flex items-center gap-1.5">
                <Users aria-hidden className="size-4 text-gold/70" />
                Up to {room.capacity} guests
              </li>
              <li className="flex items-center gap-1.5">
                <Clock aria-hidden className="size-4 text-gold/70" />
                {room.minimumDuration} min minimum
              </li>
              <li className="flex items-center gap-1.5">
                <Sparkles aria-hidden className="size-4 text-gold/70" />
                {formatMoney(room.hourlyRate)} per hour
              </li>
            </ul>
          </header>

          <section className="mt-7">
            <h2 className="text-sm font-semibold uppercase tracking-[0.1em] text-faint">
              About this room
            </h2>
            <p className="mt-3 whitespace-pre-line text-[14.5px] leading-relaxed text-muted">
              {room.description ??
                room.shortDescription ??
                "A private, soundproofed room with full table service."}
            </p>
          </section>

          {room.amenities.length > 0 ? (
            <section className="mt-7">
              <h2 className="text-sm font-semibold uppercase tracking-[0.1em] text-faint">
                Facilities &amp; amenities
              </h2>
              <ul className="mt-3 grid grid-cols-1 gap-2 sm:grid-cols-2">
                {room.amenities.map((amenity) => (
                  <li
                    key={amenity}
                    className="flex items-center gap-2 rounded-md border border-line bg-card/60 px-3 py-2.5 text-[13px] text-muted"
                  >
                    <Check aria-hidden className="size-3.5 shrink-0 text-gold" />
                    {amenity}
                  </li>
                ))}
              </ul>
            </section>
          ) : null}

          <section className="mt-7 rounded-lg border border-line bg-card p-5">
            <h2 className="text-sm font-semibold uppercase tracking-[0.1em] text-faint">
              Opening hours
            </h2>
            <p className="mt-2 text-[14px]">
              {formatOperatingDays(settings.operatingDays, DAY_LABELS)} ·{" "}
              {settings.openingTime} – {settings.closingTime}{" "}
              <span className="text-muted">({settings.timezone})</span>
            </p>
            <p className="mt-1.5 text-[12.5px] leading-relaxed text-muted">
              Bookings start on the hour or half hour. We hold your room for a limited
              window while our team confirms.
            </p>
          </section>
        </div>

        {/* Booking panel */}
        <aside className="lg:sticky lg:top-24 lg:self-start">
          <div className="rounded-lg border border-line bg-card p-5 sm:p-6">
            <p className="text-[11px] font-semibold uppercase tracking-[0.12em] text-faint">
              Hourly rate
            </p>
            <p className="mt-1 flex items-baseline gap-1.5">
              <span className="text-3xl font-semibold tracking-tight">
                {formatMoney(room.hourlyRate)}
              </span>
              <span className="text-[12.5px] text-muted">/ hour</span>
            </p>
            <p className="mt-1 text-[12px] text-faint">
              You pay at the club — no online payment.
            </p>

            <div className="mt-5 rounded-md border border-line bg-surface p-3.5">
              <p className="text-[11.5px] font-semibold uppercase tracking-[0.1em] text-faint">
                Next available today
              </p>
              {slots.length > 0 ? (
                <ul className="mt-2 flex flex-wrap gap-1.5">
                  {slots.map((slot) => (
                    <li key={slot}>
                      <Link
                        href={`/reserve?room=${room.slug}&date=${today}&time=${slot}`}
                        className="inline-block rounded-md border border-gold/30 bg-gold/10 px-2.5 py-1.5 text-[12px] font-medium text-gold transition-colors hover:bg-gold/20"
                      >
                        {formatClubTime12h(slot)}
                      </Link>
                    </li>
                  ))}
                </ul>
              ) : (
                <p className="mt-2 text-[12.5px] leading-relaxed text-muted">
                  No slots left today.{" "}
                  <Link href="/reserve" className="text-gold underline-offset-4 hover:underline">
                    Try another date
                  </Link>
                  .
                </p>
              )}
              <p className="mt-2.5 text-[11px] text-faint">
                {formatClubDateLong(today)}
              </p>
            </div>

            <Button asChild size="lg" className="mt-4 w-full">
              <Link href={`/reserve?room=${room.slug}`}>
                <CalendarCheck className="size-4" />
                Reserve this room
              </Link>
            </Button>

            <p className="mt-3 text-center text-[11.5px] leading-relaxed text-faint">
              Minimum {room.minimumDuration} minutes · Instant reference number
            </p>
          </div>

          {alternatives.length > 0 ? (
            <div className="mt-6">
              <h2 className="text-[11px] font-semibold uppercase tracking-[0.12em] text-faint">
                Other rooms
              </h2>
              <ul className="mt-3 space-y-2.5">
                {alternatives.map((candidate) => (
                  <li key={candidate.id}>
                    <Link
                      href={`/rooms/${candidate.slug}`}
                      className="flex items-center gap-3 rounded-md border border-line bg-card p-2.5 transition-colors hover:border-line-strong"
                    >
                      {/* eslint-disable-next-line @next/next/no-img-element */}
                      <img
                        src={candidate.coverImage ?? candidate.images[0] ?? "/images/room-standard.jpg"}
                        alt=""
                        className="size-14 shrink-0 rounded object-cover"
                      />
                      <span className="min-w-0 flex-1">
                        <span className="block truncate text-[13px] font-medium">
                          {candidate.name}
                        </span>
                        <span className="block text-[11.5px] text-faint">
                          {formatMoney(candidate.hourlyRate)} / hour · {candidate.capacity} guests
                        </span>
                      </span>
                      <ArrowRight aria-hidden className="size-3.5 shrink-0 text-muted" />
                    </Link>
                  </li>
                ))}
              </ul>
            </div>
          ) : null}
        </aside>
      </div>

      {similar.length > 0 ? (
        <section className="mt-14">
          <h2 className="flex items-center gap-2 font-display text-2xl tracking-tight">
            <Music4 aria-hidden className="size-5 text-gold" />
            More {categoryLabel(room.categorySlug)} rooms
          </h2>
          <div className="mt-4 grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
            {similar.map((candidate) => (
              <RoomCard key={candidate.id} room={candidate} />
            ))}
          </div>
        </section>
      ) : null}
    </div>
  );
}
